cmake_minimum_required(VERSION 3.22)
project(grabador_whisper LANGUAGES C CXX)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory(whisper-src)
add_library(grabador_whisper SHARED whisper_jni.cpp)
target_include_directories(grabador_whisper PRIVATE whisper-src/include)
target_link_libraries(grabador_whisper PRIVATE whisper log)
